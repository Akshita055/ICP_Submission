1382. Balance a Binary Search Tree

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    // to get the values in the sorted order
    public void inorder(TreeNode node, List<Integer> val){
        if(node==null) return;
        inorder(node.left,val);
        val.add(node.val);
        inorder(node.right,val);
    }dvvvb
    public TreeNode build(List<Integer> val,int left,int right){
        if(left>right) return null;
        int mid = (left+right)/2;
        TreeNode node = new TreeNode(val.get(mid));
        node.left = build(val,left,mid-1);
        node.right=build(val,mid+1,right);
        return node;
    }
    public TreeNode balanceBST(TreeNode root) {
        List<Integer> val = new ArrayList<>();
        inorder(root,val);
        return build(val,0,val.size()-1);

    }
}
