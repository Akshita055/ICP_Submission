2673. Make Costs of Paths Equal in a Binary Tree
Medium

class Solution {
    public int minIncrements(int n, int[] cost) {
        int ans = 0;
        for(int i = n/2;i>=1;i--){
            int l = cost[2*i-1];
            int r = cost[2*i];

            ans+= Math.abs(l-r);

            cost[i-1]+=Math.max(l,r);
        }
        return ans;
    }
}
